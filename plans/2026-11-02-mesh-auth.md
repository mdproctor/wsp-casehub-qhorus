**Focal issue:** #508 — feat: mesh service auth — OIDC integration + API key fallback
**Issue group:** #508

**Goal:** Add OIDC and API key authentication to the standalone mesh service so that all endpoints require authenticated identity.

**Architecture:** Custom `HttpAuthenticationMechanism` reads `X-Api-Key` header and delegates to an `ApiKeyIdentityResolver` SPI. A `@DefaultBean` config-based resolver maps keys to actorIds (single-tenant). Adding `casehub-platform-oidc` as a dependency auto-enables OIDC via CDI `@Alternative @Priority(100)` displacement. Both mechanisms coexist — OIDC tried first, API key as fallback.

**Tech Stack:** Java 21, Quarkus 3.32.2, Quarkus Security (`HttpAuthenticationMechanism`, `SecurityIdentity`), SmallRye Config (`@ConfigMapping`)

## Global Constraints

- SPI interface goes in `api/spi/` (same pattern as all qhorus SPIs)
- `@DefaultBean` impl goes in `runtime-core/` (same pattern as `DefaultMeshAutoRegistrar`)
- `HttpAuthenticationMechanism` goes in `mesh/` (standalone-service concern)
- Config prefix: `casehub.qhorus.api-keys.*`
- No database changes, no Flyway migrations
- No endpoint-level authorization (`@RolesAllowed`) — authentication only

```java
package io.casehub.qhorus.runtime.identity;

import io.casehub.platform.api.identity.TenancyConstants;
import io.casehub.qhorus.api.spi.ApiKeyIdentityResolver;
import io.casehub.qhorus.api.spi.ResolvedIdentity;
import io.casehub.qhorus.runtime.config.QhorusConfig;
import io.quarkus.arc.DefaultBean;
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;

import java.util.Map;
import java.util.Optional;
import java.util.Set;

@DefaultBean
@ApplicationScoped
public class ConfigApiKeyIdentityResolver implements ApiKeyIdentityResolver {

    private final Map<String, KeyEntry> keys;
    private final String tenancyId;

    public record KeyEntry(String key, Optional<Set<String>> roles) {}

    @Inject
    public ConfigApiKeyIdentityResolver(QhorusConfig config) {
        this(buildKeyMap(config), config.defaultTenancyId().orElse(TenancyConstants.DEFAULT_TENANT_ID));
    }

    ConfigApiKeyIdentityResolver(Map<String, KeyEntry> keys, String tenancyId) {
        this.keys = keys;
        this.tenancyId = tenancyId;
    }

    @Override
    public Optional<ResolvedIdentity> resolve(String apiKey) {
        return keys.entrySet().stream()
            .filter(e -> e.getValue().key().equals(apiKey))
            .findFirst()
            .map(e -> new ResolvedIdentity(
                e.getKey(),
                tenancyId,
                e.getValue().roles().orElse(Set.of())));
    }

    private static Map<String, KeyEntry> buildKeyMap(QhorusConfig config) {
        var configKeys = config.apiKeys();
        if (configKeys == null) return Map.of();
        var builder = new java.util.HashMap<String, KeyEntry>();
        configKeys.forEach((name, entry) ->
            builder.put(name, new KeyEntry(entry.key(), entry.roles())));
        return Map.copyOf(builder);
    }
}
```

```java
package io.casehub.qhorus.mesh.auth;

import io.casehub.platform.api.identity.SecurityIdentityAttributes;
import io.casehub.qhorus.api.spi.ApiKeyIdentityResolver;
import io.casehub.qhorus.api.spi.ResolvedIdentity;
import io.quarkus.security.identity.IdentityProviderManager;
import io.quarkus.security.identity.SecurityIdentity;
import io.quarkus.security.runtime.QuarkusSecurityIdentity;
import io.smallrye.mutiny.Uni;
import io.vertx.ext.web.RoutingContext;
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;

import java.security.Principal;

@ApplicationScoped
public class ApiKeyAuthenticationMechanism
        implements io.quarkus.vertx.http.runtime.security.HttpAuthenticationMechanism {

    private final ApiKeyIdentityResolver resolver;

    @Inject
    public ApiKeyAuthenticationMechanism(ApiKeyIdentityResolver resolver) {
        this.resolver = resolver;
    }

    @Override
    public Uni<SecurityIdentity> authenticate(
            RoutingContext context,
            IdentityProviderManager identityProviderManager) {
        String key = context.request().getHeader("X-Api-Key");
        if (key == null || key.isBlank()) {
            return Uni.createFrom().nullItem();
        }
        return Uni.createFrom().item(() ->
            resolver.resolve(key).map(this::toSecurityIdentity).orElse(null));
    }

    @Override
    public int getPriority() {
        return 2000;
    }

    private SecurityIdentity toSecurityIdentity(ResolvedIdentity resolved) {
        return QuarkusSecurityIdentity.builder()
            .setPrincipal(new ApiKeyPrincipal(resolved.actorId()))
            .addRoles(resolved.roles())
            .addAttribute(SecurityIdentityAttributes.TENANCY_ID, resolved.tenancyId())
            .build();
    }

    private record ApiKeyPrincipal(String name) implements Principal {
        @Override public String getName() { return name; }
    }
}
```
